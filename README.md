# node

v22+

# Eslint

npm install -D eslint @eslint/js typescript typescript-eslint eslint-plugin-vue

# 项目结构

src/
├── api/ # 接口请求
├── assets/ # 静态资源
├── components/ # 公共组件
├── hooks/ # 组合式函数
├── layouts/ # 布局组件
├── pages/ # 页面组件
├── router/ # 路由配置
├── services/ # axios服务层
├── stores/ # Pinia 状态管理
│ ├── global.ts # 全局状态管理
│ ├── auth.ts # 用户状态管理
├── styles/ # 全局样式
├── types/ # TS 类型
├── utils/ # 工具函数
├── App.vue # 入口组件
└── main.ts # 入口文件

# 安装 Node 类型：

npm install -D @types/node

# Vue Router

npm install vue-router@latest

# Pinia

npm install pinia@latest
npm i pinia-plugin-persistedstate 安装持久化插件

# 安装 scss

npm install -D sass

# 安装滚动条

npm install vue3-perfect-scrollbar

# 安装UI框架 NaiveUI

npm install naive-ui

# 安装 ionicons5 图标

npm i -D @vicons/ionicons5

# 安装 SVG Sprite，用于 SVG 图标

npm i vite-plugin-svg-icons -D

# 配置自动引入

npm install -D unplugin-auto-import
npm install -D @unplugin-vue-components

# 安装 axios 封装为装饰器请求函数

npm install axios
npm install reflect-metadata 用于装饰器元数据
